cmake_minimum_required(VERSION 3.8)
project(mobile_openarm_shared_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(dwb_core REQUIRED)
find_package(dwb_msgs REQUIRED)
find_package(nav_2d_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/HumanIntent.msg")
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_library(human_influence_critic SHARED src/human_influence_critic.cpp)
target_include_directories(human_influence_critic PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(human_influence_critic rclcpp rcl_interfaces dwb_core dwb_msgs nav_2d_msgs pluginlib std_msgs)
target_link_libraries(human_influence_critic "${typesupport}")
pluginlib_export_plugin_description_file(dwb_core critics.xml)
install(TARGETS human_influence_critic LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(PROGRAMS scripts/shared_control_node.py DESTINATION lib/${PROJECT_NAME} RENAME shared_control_node)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_preference test/test_preference.cpp)
  target_include_directories(test_preference PRIVATE include)
  ament_add_pytest_test(test_gate test/test_gate.py TIMEOUT 60)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
